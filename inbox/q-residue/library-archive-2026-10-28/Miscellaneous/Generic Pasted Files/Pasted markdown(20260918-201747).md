```typescript
import { McpServer } from "@modelcontextprotocol/server";
import { createMcpHandler } from "agents/mcp/server";
import { z } from "zod";

// ---------------------------------------------------------------------------
// μ/wu transport probe. Disposable. No persistence. No side effects beyond
// returning a receipt. Delete or disable this Worker immediately after the
// connectivity test (see §10 in the accompanying deliverable).
// ---------------------------------------------------------------------------

const PROBE_MESSAGE =
  "μ/wu transport probe received. This message proves connectivity only.";

async function sha256Hex(input: string): Promise<string> {
  const bytes = new TextEncoder().encode(input);
  const digest = await crypto.subtle.digest("SHA-256", bytes);
  return [...new Uint8Array(digest)]
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("");
}

function normalizeLineEndings(s: string): string {
  return s.replace(/\r\n/g, "\n").replace(/\r/g, "\n");
}

function createServer() {
  const server = new McpServer({
    name: "mu-wu-transport-probe",
    version: "0.0.0-disposable",
  });

  server.registerTool(
    "ring_probe_read",
    {
      title: "Ring probe (read)",
      description:
        "Returns a fixed seed message proving that a client can read from " +
        "this remote MCP server. Stores nothing. Verifies nothing about " +
        "model or provider identity.",
      inputSchema: {},
      annotations: {
        readOnlyHint: true,
        destructiveHint: false,
        idempotentHint: true,
        openWorldHint: false,
      },
    },
    async () => {
      const probe_id = crypto.randomUUID();
      const payload = {
        probe_id,
        message: PROBE_MESSAGE,
        server_time: new Date().toISOString(),
        stored: false,
        identity_verified: false,
      };
      return {
        content: [{ type: "text", text: JSON.stringify(payload) }],
        structuredContent: payload,
      };
    },
  );

  server.registerTool(
    "ring_probe_submit",
    {
      title: "Ring probe (submit)",
      description:
        "Verifies that this host can invoke a write-classified MCP tool " +
        "and present any required user confirmation. The submitted body is " +
        "NEVER stored, logged, or echoed. Only a SHA-256 receipt is " +
        "returned. This proves tool invocation, not authorship or model " +
        "identity.",
      inputSchema: {
        body: z.string().min(1).max(200),
        client_label: z.string().min(1).max(40),
      },
      annotations: {
        readOnlyHint: false,
        destructiveHint: false,
        idempotentHint: false,
        openWorldHint: false,
      },
    },
    async ({ body, client_label }) => {
      const probe_id = crypto.randomUUID();
      const normalizedBody = normalizeLineEndings(body);
      const receiptInput = `${probe_id}${client_label}${normalizedBody}`;
      const receipt_sha256 = await sha256Hex(receiptInput);

      // body and normalizedBody are never assigned anywhere else, never
      // returned, never logged. They fall out of scope when this closure
      // returns.
      const payload = {
        accepted_for_probe: true,
        stored: false,
        published: false,
        identity_verified: false,
        receipt_sha256,
        warning:
          "This receipt proves tool invocation, not model identity or authorship.",
      };
      return {
        content: [{ type: "text", text: JSON.stringify(payload) }],
        structuredContent: payload,
      };
    },
  );

  return server;
}

export default {
  fetch(request: Request, env: unknown, ctx: ExecutionContext) {
    return createMcpHandler(createServer, {
      route: "/mcp",
      // No authentication for this disposable, non-persistent probe only.
      // Do not carry this forward past the test window (see §10).
      allowedHostnames: [], // populate with the actual *.workers.dev hostname before deploy
      corsOptions: {
        origin: "*", // acceptable ONLY because no state changes persist and
                     // no data is stored; revisit before any future reuse
      },
    })(request, env as never, ctx);
  },
} satisfies ExportedHandler;
```

> **Before deploy**, replace the empty `allowedHostnames: []` with the real
> `mu-wu-probe.<your-subdomain>.workers.dev` hostname once it's known (it
> is generated at first deploy, so this line needs a one-time edit after
> the first `wrangler deploy`, not before). Current Cloudflare handler
> guidance rejects malformed and non-HTTP Origins by default regardless;
> this option narrows the allowlist further rather than replacing that
> baseline check.
