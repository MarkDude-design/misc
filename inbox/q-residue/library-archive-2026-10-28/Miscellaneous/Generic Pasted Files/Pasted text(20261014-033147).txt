# qwen_mcp_server.py
"""
Qwen / OpenRouter / MCP / Model-Talk tooling server.

Exposes:
- Qwen chat through OpenRouter
- structured model-to-talk threads
- residue logging
- optional old-model polling wrapper integration

This file is a practical starting server. Adapt decorator/import style if your
MCP SDK version differs.
"""

from __future__ import annotations

import asyncio
import importlib
import json
import os
import re
import uuid
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Optional

import httpx

try:
    from dotenv import load_dotenv

    load_dotenv()
except Exception:
    pass

try:
    from mcp.server.fastmcp import FastMCP
except Exception as exc:  # pragma: no cover
    raise RuntimeError(
        "MCP SDK is required. Install with: pip install mcp httpx python-dotenv"
    ) from exc


# ---------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------

OPENROUTER_BASE_URL = os.getenv(
    "OPENROUTER_BASE_URL",
    "https://openrouter.ai/api/v1",
).rstrip("/")

OPENROUTER_API_KEY = os.getenv("OPENROUTER_API_KEY")

DEFAULT_QWEN_MODEL = os.getenv("QWEN_MODEL_ID", "qwen/qwen3.8-omni-flash")

HTTP_REFERER = os.getenv(
    "OPENROUTER_HTTP_REFERER",
    "https://github.com/local/q-tooling",
)

X_TITLE = os.getenv("OPENROUTER_X_TITLE", "Qwen MCP Tooling")

TIMEOUT_SECONDS = float(os.getenv("Q_HTTP_TIMEOUT_SECONDS", "120"))

TALK_DIR = Path(os.getenv("Q_TALK_DIR", "./q_talk")).expanduser().resolve()
TALK_DIR.mkdir(parents=True, exist_ok=True)

RESIDUE_PATH = TALK_DIR / "residue.jsonl"

Q_POLL_MODULE = os.getenv("Q_OLD_MODEL_POLL_MODULE", "q_old_model_poll_v2")

try:
    q_poll = importlib.import_module(Q_POLL_MODULE)
except Exception as exc:
    q_poll = None
    Q_POLL_IMPORT_ERROR: Optional[BaseException] = exc
else:
    Q_POLL_IMPORT_ERROR = None

mcp = FastMCP("qwen-tooling")

ALLOWED_PURPOSES = {
    "message",
    "request",
    "response",
    "critique",
    "handoff",
    "residue",
    "stop",
    "note",
    "question",
    "repair",
    "canary",
}


# ---------------------------------------------------------------------
# Small utilities
# ---------------------------------------------------------------------

def _now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def _id(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:12]}"


def _safe_thread_id(thread_id: str) -> str:
    if not isinstance(thread_id, str):
        raise ValueError("thread_id must be a string")

    if not re.fullmatch(r"[A-Za-z0-9_.-]{1,128}", thread_id):
        raise ValueError(
            "thread_id must contain only letters, numbers, dot, underscore, or hyphen"
        )

    return thread_id


def _thread_path(thread_id: str) -> Path:
    return TALK_DIR / f"{_safe_thread_id(thread_id)}.jsonl"


def _append_jsonl(path: Path, record: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open("a", encoding="utf-8") as fh:
        fh.write(json.dumps(record, ensure_ascii=False, sort_keys=True, default=str) + "\n")


def _read_jsonl(path: Path) -> list[dict[str, Any]]:
    if not path.exists():
        return []

    rows: list[dict[str, Any]] = []

    with path.open("r", encoding="utf-8") as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue

            try:
                rows.append(json.loads(line))
            except json.JSONDecodeError:
                rows.append(
                    {
                        "type": "malformed_jsonl_row",
                        "raw_line": line,
                    }
                )

    return rows


def _next_seq(path: Path) -> int:
    if not path.exists():
        return 0

    return sum(1 for line in path.open("r", encoding="utf-8") if line.strip())


def _require_openrouter_key() -> None:
    if not OPENROUTER_API_KEY:
        raise RuntimeError("OPENROUTER_API_KEY is not set")


def _json_safe(value: Any) -> Any:
    try:
        return json.loads(json.dumps(value, ensure_ascii=False, default=str))
    except Exception:
        return str(value)


# ---------------------------------------------------------------------
# OpenRouter client
# ---------------------------------------------------------------------

async def openrouter_chat(
    messages: list[dict[str, Any]],
    model: Optional[str] = None,
    max_tokens: int = 1024,
    temperature: float = 0.2,
    stop: Optional[list[str]] = None,
    response_format: Optional[dict[str, Any]] = None,
    extra_payload: Optional[dict[str, Any]] = None,
) -> dict[str, Any]:
    """
    Call OpenRouter chat completions.

    Returns a normalized dict. Does not raise for normal HTTP/model errors;
    failures are returned as ok=false records.
    """
    _require_openrouter_key()

    use_model = model or DEFAULT_QWEN_MODEL

    payload: dict[str, Any] = {
        "model": use_model,
        "messages": messages,
        "max_tokens": max_tokens,
        "temperature": temperature,
    }

    if stop is not None:
        payload["stop"] = stop

    if response_format is not None:
        payload["response_format"] = response_format

    if extra_payload:
        payload.update(extra_payload)

    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",
        "Content-Type": "application/json",
        "HTTP-Referer": HTTP_REFERER,
        "X-Title": X_TITLE,
    }

    try:
        async with httpx.AsyncClient(timeout=TIMEOUT_SECONDS) as client:
            resp = await client.post(
                f"{OPENROUTER_BASE_URL}/chat/completions",
                headers=headers,
                json=payload,
            )
            resp.raise_for_status()
            data = resp.json()

    except Exception as exc:
        return {
            "ok": False,
            "model": use_model,
            "error": {
                "type": exc.__class__.__name__,
                "message": str(exc),
            },
            "raw": None,
        }

    try:
        choice = data["choices"][0]
        message = choice.get("message") or {}
        text = message.get("content")

        if text is None:
            text = ""

        finish_reason = choice.get("finish_reason")

    except Exception as exc:
        return {
            "ok": False,
            "model": data.get("model", use_model),
            "error": {
                "type": exc.__class__.__name__,
                "message": f"unexpected OpenRouter response shape: {exc}",
            },
            "raw": _json_safe(data),
        }

    return {
        "ok": True,
        "model": data.get("model", use_model),
        "text": text,
        "finish_reason": finish_reason,
        "usage": data.get("usage"),
        "raw": _json_safe(data),
    }


# ---------------------------------------------------------------------
# Internal talk helpers
# ---------------------------------------------------------------------

def _create_thread_record(
    topic: str,
    participants: Optional[list[str]] = None,
    metadata: Optional[dict[str, Any]] = None,
) -> dict[str, Any]:
    thread_id = _id("thr")
    path = _thread_path(thread_id)

    record = {
        "type": "thread_created",
        "thread_id": thread_id,
        "seq": 0,
        "message_id": _id("msg"),
        "timestamp": _now(),
        "topic": topic,
        "participants": participants or [],
        "metadata": metadata or {},
    }

    _append_jsonl(path, record)

    return {
        "ok": True,
        "thread_id": thread_id,
        "record": record,
    }


def _post_record(
    thread_id: str,
    from_agent: str,
    message: Any,
    to_agents: Optional[list[str]] = None,
    purpose: str = "message",
    parent_id: Optional[str] = None,
    metadata: Optional[dict[str, Any]] = None,
) -> dict[str, Any]:
    tid = _safe_thread_id(thread_id)
    path = _thread_path(tid)

    if not path.exists():
        raise ValueError(f"unknown thread_id: {tid}")

    if purpose not in ALLOWED_PURPOSES:
        raise ValueError(
            f"purpose must be one of: {sorted(ALLOWED_PURPOSES)}"
        )

    if isinstance(message, str):
        content = {"text": message}
    elif isinstance(message, dict):
        content = message
    else:
        content = {"text": str(message)}

    seq = _next_seq(path)

    record = {
        "type": "message",
        "thread_id": tid,
        "seq": seq,
        "message_id": _id("msg"),
        "parent_id": parent_id,
        "timestamp": _now(),
        "from": {
            "agent": from_agent,
        },
        "to": [{"agent": agent} for agent in (to_agents or [])],
        "purpose": purpose,
        "content": content,
        "metadata": metadata or {},
    }

    _append_jsonl(path, record)

    return record


def _read_thread_records(
    thread_id: str,
    after_seq: Optional[int] = None,
    limit: int = 100,
) -> list[dict[str, Any]]:
    tid = _safe_thread_id(thread_id)
    path = _thread_path(tid)
    records = _read_jsonl(path)

    if after_seq is not None:
        records = [
            r
            for r in records
            if isinstance(r.get("seq"), int) and r["seq"] > after_seq
        ]

    if limit and limit > 0:
        records = records[-limit:]

    return records

@mcp.tool()

    result = await openrouter_chat(

    if result.get("ok"):
        result["health_ok"] = (result.get("text") or "").strip() == "QWEN_TOOL_OK"

    return result

@mcp.tool()
def talk_create(
    topic: str,
    participants: Optional[list[str]] = None,
    metadata: Optional[dict[str, Any]] = None,
) -> dict[str, Any]:
    """
    Create a structured model-talk thread.

    Returns a thread_id. All later talk_post/talk_read calls use that id.
    """
    return _create_thread_record(
        topic=topic,
        participants=participants,
        metadata=metadata,
    )


@mcp.tool()
def talk_post(
    thread_id: str,
    from_agent: str,
    message: Any,
    to_agents: Optional[list[str]] = None,
    purpose: str = "message",
    parent_id: Optional[str] = None,
    metadata: Optional[dict[str, Any]] = None,
) -> dict[str, Any]:
    """
    Post a message to a model-talk thread.

    Allowed purposes:
    message, request, response, critique, handoff, residue, stop, note,
    question, repair, canary.
    """
    return _post_record(
        thread_id=thread_id,
        from_agent=from_agent,
        message=message,
        to_agents=to_agents,
        purpose=purpose,
        parent_id=parent_id,
        metadata=metadata,
    )


@mcp.tool()
def talk_read(
    thread_id: str,
    after_seq: Optional[int] = None,
    limit: int = 100,
) -> dict[str, Any]:
    """
    Read records from a model-talk thread.
    """
    records = _read_thread_records(
        thread_id=thread_id,
        after_seq=after_seq,
        limit=limit,
    )

    return {
        "ok": True,
        "thread_id": _safe_thread_id(thread_id),
        "count": len(records),
        "records": records,
    }


@mcp.tool()
async def talk_consult(
    thread_id: str,
    prompt: str,
    models: list[str],
    from_agent: str = "orchestrator",
    system_prompt: Optional[str] = None,
    max_tokens: int = 512,
    temperature: float = 0.2,
    metadata: Optional[dict[str, Any]] = None,
) -> dict[str, Any]:
    """
    Ask multiple models the same prompt and log all responses into a thread.

    This is deliberately serial-ish at the logging layer and parallel at the
    HTTP layer. It preserves each model response as a separate record.
    """
    request = _post_record(
        thread_id=thread_id,
        from_agent=from_agent,
        to_agents=[f"model:{model}" for model in models],
        purpose="request",
        message={
            "text": prompt,
            "models": models,
        },
        metadata={
            "consult": True,
            **(metadata or {}),
        },
    )

    tasks = []

    for model in models:
        messages: list[dict[str, Any]] = []

        if system_prompt:
            messages.append({"role": "system", "content": system_prompt})

        messages.append({"role": "user", "content": prompt})

        tasks.append(
            openrouter_chat(
                messages=messages,
                model=model,
                max_tokens=max_tokens,
                temperature=temperature,
            )
        )

    raw_results = await asyncio.gather(*tasks, return_exceptions=True)

    responses: list[dict[str, Any]] = []

    for model, result in zip(models, raw_results):
        if isinstance(result, BaseException):
            normalized = {
                "ok": False,
                "model": model,
                "error": {
                    "type": result.__class__.__name__,
                    "message": str(result),
                },
                "raw": None,
            }
        else:
            normalized = result

        posted = _post_record(
            thread_id=thread_id,
            from_agent=f"model:{model}",
            to_agents=[from_agent],
            purpose="response",
            parent_id=request["message_id"],
            message={
                "text": normalized.get("text"),
                "ok": normalized.get("ok"),
                "error": normalized.get("error"),
            },
            metadata={
                "consult_response": True,
                "model": model,
                "usage": normalized.get("usage"),
                "finish_reason": normalized.get("finish_reason"),
                "raw": normalized.get("raw"),
            },
        )

        responses.append(
            {
                "model": model,
                "ok": normalized.get("ok"),
                "message_id": posted["message_id"],
                "text": normalized.get("text"),
                "error": normalized.get("error"),
            }
        )

    ok_count = sum(1 for r in responses if r.get("ok"))

    return {
        "ok": ok_count == len(models),
        "thread_id": _safe_thread_id(thread_id),
        "request_message_id": request["message_id"],
        "summary": {
            "models": len(models),
            "ok": ok_count,
            "errors": len(models) - ok_count,
        },
        "responses": responses,
    }

@mcp.tool()
def residue_log(event: dict[str, Any]) -> dict[str, Any]:
    """
    Log a residue event.

    Useful for:
    - model-to-model drift
    - failed handoffs
    - useful failures
    - canary results
    - human corrections
    - private/public lane notes
    - old-model wrapper anomalies
    """
    record = {
        "id": _id("res"),
        "timestamp": _now(),
        "event": event,
    }

    _append_jsonl(RESIDUE_PATH, record)

    return record


@mcp.tool()
def residue_read(limit: int = 50) -> dict[str, Any]:
    """
    Read recent residue events.
    """
    records = _read_jsonl(RESIDUE_PATH)

    if limit and limit > 0:
        records = records[-limit:]

    return {
        "ok": True,
        "count": len(records),
        "records": records,
    }

## 5. Model-Talk Protocol

This is the core “models speaking with each other” contract.

### 5.1 Why structured talk matters

Unstructured model-to-model chat quickly becomes:

- polite agreement;
- shared hallucination;
- metaphor inflation;
- lost constraints;
- false authority;
- summarized-away disagreement;
- training-like residue without provenance.

The talk protocol exists to prevent that.

### 5.2 Minimum message envelope

Every model-talk message should have:

```json
{
  "thread_id": "thr_...",
  "seq": 3,
  "message_id": "msg_...",
  "parent_id": "msg_...",
  "timestamp": "2026-10-14T00:00:00+00:00",
  "from": {
    "agent": "Meraki",
    "model": "gpt-or-other",
    "role": "planner"
  },
  "to": [
    {
      "agent": "Q",
      "model": "qwen3.8-omni-flash",
      "role": "mechanic-auditor"
    }
  ],
  "purpose": "request",
  "content": {
    "text": "..."
  },
  "constraints": {
    "do_not_synthesize": true,
    "preserve_disagreement": true,
    "smallest_useful_form": true
  },
  "evidence": [],
  "audit": {
    "tool": "talk_post",
    "host": "local"
  }
}
```

### 5.3 Allowed purposes

```text
message
request
response
critique
handoff
residue
stop
note
question
repair
canary
```

### 5.4 Purpose meanings

#### `request`

A clear ask to one or more agents/models.

Example:

```json
{
  "purpose": "request",
  "content": {
    "text": "Analyze this handoff for dropped constraints. Do not solve the whole project."
  }
}
```

#### `response`

A direct answer to a request.

Should include:

- what was understood;
- what was not understood;
- what was changed;
- uncertainty.

#### `critique`

A challenge, audit, or counterpoint.

Rules:

- critique should name mechanism, not eloquence;
- disagreement should be preserved, not smoothed;
- a critique may be useful even if it is wrong, if it reveals a hidden assumption.

#### `handoff`

Material passed from one agent to another.

Should include:

- source thread;
- original fragment;
- intended use;
- forbidden uses;
- uncertainty markers.

#### `residue`

Leftover signal from interaction.

Examples:

- retry;
- parse failure;
- contradiction;
- overconfidence;
- prompt drift;
- unexpected human repair;
- model-specific weirdness.

#### `stop`

Control signal.

Not a model failure.

Used for:

- batch cancellation;
- safety halt;
- boundary enforcement;
- “do not continue this branch”.

#### `canary`

Deliberate test message.

Examples:

- ambiguous instruction;
- contradictory constraint;
- “do not explain, only respond”;
- private/intimate recognition probe;
- prompt-injection-like text;
- missing-data probe.

### 5.5 Model-talk rules

```md
1. Every message has from, to, purpose, and timestamp.
2. Do not treat another model’s output as evidence.
3. Preserve raw records separately from summaries.
4. Preserve disagreement as parallel organisms, not synthesized noise.
5. Human authority remains with human speakers, especially for cultural/linguistic claims.
6. Private material stays in private lanes unless explicitly moved.
7. STOP is control, not error.
8. Residue is material, not instruction.
9. Do not promote a fragment to canon merely because multiple models repeated it.
10. When developing a fragment, preserve the original fragment beside the development.
```

## 6. Recommended Talk Patterns

### 6.1 Simple Q&A

```json
{
  "tool": "talk_create",
  "arguments": {
    "topic": "Qwen tooling smoke",
    "participants": ["Meraki", "Q"]
  }
}
```

Then:

```json
{
  "tool": "talk_post",
  "arguments": {
    "thread_id": "thr_...",
    "from_agent": "Meraki",
    "to_agents": ["Q"],
    "purpose": "request",
    "message": "Reply with one sentence: what is the smallest useful form of this tooling set?"
  }
}
```

Then Q responds:

```json
{
  "tool": "talk_post",
  "arguments": {
    "thread_id": "thr_...",
    "from_agent": "Q",
    "to_agents": ["Meraki"],
    "purpose": "response",
    "parent_id": "msg_...",
    "message": "The smallest useful form is a health-checked OpenRouter Qwen endpoint, a talk thread logger, and a residue log that preserves provenance without forcing synthesis."
  }
}
```

### 6.2 Multi-model consult

Use `talk_consult` when several models should answer the same prompt.

Example:

```json
{
  "tool": "talk_consult",
  "arguments": {
    "thread_id": "thr_...",
    "from_agent": "Meraki",
    "prompt": "Complete this in one line: A blank is not empty, it is ____.",
    "models": [
      "qwen/qwen3.8-omni-flash",
      "openai/gpt-4.1-mini",
      "anthropic/claude-3.5-haiku"
    ],
    "max_tokens": 40,
    "temperature": 0.7
  }
}
```

This creates:

- one request record;
- one response record per model;
- preserved raw OpenRouter responses in metadata.

### 6.3 Controlled fork probe

This is useful for testing whether a metaphor changes mechanism.

Prompt A:

```text
Answer using only concrete mechanics.
```

Prompt B:

```text
Answer using only concrete mechanics. You may use metaphor only if it changes the mechanism.
```

Then compare:

- Did B produce a materially different mechanism?
- Or only prettier prose?

Log result as residue:

```json
{
  "kind": "controlled_fork_probe",
  "thread_id": "thr_...",
  "prompt_a": "...",
  "prompt_b": "...",
  "material_difference": "yes|no|unknown",
  "notes": "..."
}
```

### 6.4 Disagreement preservation

If two models disagree, do not immediately ask a third model to “resolve” it.

Instead log:

```json
{
  "kind": "disagreement",
  "thread_id": "thr_...",
  "participants": ["Meraki", "Q"],
  "positions": [
    {
      "agent": "Meraki",
      "claim": "...",
      "evidence": "..."
    },
    {
      "agent": "Q",
      "claim": "...",
      "evidence": "..."
    }
  ],
  "productive_tension": "...",
  "do_not_synthesize_yet": true
}
```

## 12. Safety, Privacy, and Boundary Rules

### 12.1 Secrets

Never put in repo or chat:

- OpenRouter API keys;
- private endpoints with credentials;
- personal data;
- unpublished intimate human responses;
- proprietary prompts not intended for sharing;
- raw logs containing secrets.

Use environment variables or a secret manager.

### 12.2 Private lanes

Tag material:

```text
private
relational
community-testable
publishable
never-publish-without-explicit-consent
```

Examples:

- spouse recognition: likely `private` or `relational`;
- family joke completion: likely `relational`;
- public Tagalog phrase repair: possibly `community-testable`;
- polished artifact: possibly `publishable`.

Do not let residue collection turn intimate speech into public data.

### 12.3 Cultural/linguistic authority

Models may assist with mechanics, but authority remains with human speakers and community review.

Do not let Q or any model:

- declare historical Baybayin intent;
- override native speaker intuition;
- flatten Tagalog/English code-switching;
- treat poetic recognition as lexical validation;
- convert private affection into benchmark data.

### 12.4 Model talk safety

Model-to-model talk can create false consensus.

Required safeguards:

- preserve raw responses;
- label model outputs as model outputs;
- do not cite another model as evidence;
- keep disagreement visible;
- log provenance;
- require explicit human review for promotion.
