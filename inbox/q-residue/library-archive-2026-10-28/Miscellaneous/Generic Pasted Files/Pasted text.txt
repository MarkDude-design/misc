<head>
    <title>PEO-01: Progressive Equilibrium Failure</title>
    <style>
        body { margin: 0; overflow: hidden; background: #0a0a1a; font-family: sans-serif; }
        canvas { display: block; }
        .controls {
            position: absolute; top: 10px; left: 10px;
            background: rgba(0,0,0,0.7); color: #eee; padding: 12px;
            border-radius: 8px; font-size: 14px;
        }
        .controls label { display: block; margin: 8px 0 4px; }
        button { padding: 6px 12px; margin: 2px; background: #333; color: #fff;
                 border: none; border-radius: 4px; cursor: pointer; }
        button:hover { background: #555; }
        #status { font-weight: bold; }
    </style>

<body>
    <canvas id="canvas"></canvas>
    <div class="controls">
        <label>Speed: <input type="range" id="speed" min="0" max="2" step="0.05" value="1"></label>
        <button id="reset">Reset</button>
        <div>Click to disturb</div>
        <div id="status">Equilibrium: STABLE</div>
    </div>
    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const speedControl = document.getElementById('speed');
        const resetBtn = document.getElementById('reset');
        const statusDisplay = document.getElementById('status');

        function resize() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        const NATURAL_LENGTH = 180;
        const SPRING_CONSTANT = 0.08;
        const DAMPING = 0.97;
        const CENTRAL_FORCE = 0.015;
        const BREAKING_THRESHOLD = NATURAL_LENGTH * 2.2;
        const ENERGY_THRESHOLD = 800;

        const bodies = [];
        let disturbanceCount = 0;
        let totalEnergy = 0;
        let animationId;
        let speed = 1;
        let systemState = 'stable';

        function reset() {
            disturbanceCount = 0;
            totalEnergy = 0;
            systemState = 'stable';
            bodies.length = 0;

            const cx = canvas.width / 2;
            const cy = canvas.height / 2;

            for (let i = 0; i < 3; i++) {
                const angle = Math.PI * 2 * i / 3 - Math.PI / 2;
                bodies.push({
                    x: cx + Math.cos(angle) * NATURAL_LENGTH,
                    y: cy + Math.sin(angle) * NATURAL_LENGTH,
                    vx: 0, vy: 0,
                    color: ['#ff6b6b', '#51cf66', '#339af0'][i],
                    radius: 20,
                    trail: [],
                    originalRadius: 20
                });
            }

            for (let i = 0; i < bodies.length; i++) {
                const dx = bodies[i].x - cx;
                const dy = bodies[i].y - cy;
                bodies[i].vx = -dy * 0.005;
                bodies[i].vy = dx * 0.005;
            }
            updateStatus();
        }

        reset();

        function applySprings() {
            for (let i = 0; i < bodies.length; i++) {
                for (let j = i + 1; j < bodies.length; j++) {
                    const a = bodies[i], b = bodies[j];
                    const dx = b.x - a.x, dy = b.y - a.y;
                    const dist = Math.sqrt(dx*dx + dy*dy);
                    if (dist === 0) continue;
                    const force = SPRING_CONSTANT * (dist - NATURAL_LENGTH);
                    const fx = (dx / dist) * force;
                    const fy = (dy / dist) * force;
                    a.vx += fx; a.vy += fy;
                    b.vx -= fx; b.vy -= fy;
                }
            }
        }

        function applyCentralForce() {
            const cx = canvas.width / 2, cy = canvas.height / 2;
            for (let i = 0; i < bodies.length; i++) {
                const b = bodies[i];
                const dx = cx - b.x, dy = cy - b.y;
                const dist = Math.sqrt(dx*dx + dy*dy);
                if (dist > 0) {
                    b.vx += (dx / dist) * CENTRAL_FORCE;
                    b.vy += (dy / dist) * CENTRAL_FORCE;
                }
            }
        }

        function disturb(x, y) {
            if (systemState === 'terminal') { reset(); return; }
            disturbanceCount++;
            const cx = canvas.width / 2, cy = canvas.height / 2;
            let closestIndex = 0, closestDist = Infinity;
            for (let i = 0; i < bodies.length; i++) {
                const dx = bodies[i].x - x, dy = bodies[i].y - y;
                const dist = Math.sqrt(dx*dx + dy*dy);
                if (dist < closestDist) { closestDist = dist; closestIndex = i; }
            }
            const body = bodies[closestIndex];
            const angle = Math.atan2(body.y - y, body.x - x);
            const impulse = 4 + disturbanceCount * 1.2;
            body.vx += Math.cos(angle) * impulse;
            body.vy += Math.sin(angle) * impulse;
            totalEnergy += impulse * impulse;
            if (disturbanceCount >= 3 && systemState === 'stable') systemState = 'endangered';
            if (totalEnergy > ENERGY_THRESHOLD * 0.7 && systemState === 'endangered') systemState = 'failing';
            updateStatus();
        }

        function checkTerminalFailure() {
            for (let i = 0; i < bodies.length; i++) {
                for (let j = i + 1; j < bodies.length; j++) {
                    const dx = bodies[j].x - bodies[i].x;
                    const dy = bodies[j].y - bodies[i].y;
                    if (Math.sqrt(dx*dx + dy*dy) > BREAKING_THRESHOLD) return true;
                }
            }
            return totalEnergy > ENERGY_THRESHOLD;
        }

        function getAverageDistance() {
            let sum = 0, count = 0;
            for (let i = 0; i < bodies.length; i++) {
                for (let j = i + 1; j < bodies.length; j++) {
                    const dx = bodies[j].x - bodies[i].x;
                    const dy = bodies[j].y - bodies[i].y;
                    sum += Math.sqrt(dx*dx + dy*dy);
                    count++;
                }
            }
            return sum / count;
        }

        function updateStatus() {
            if (checkTerminalFailure()) {
                systemState = 'terminal';
                statusDisplay.textContent = 'Equilibrium: TERMINAL FAILURE';
                statusDisplay.style.color = '#f00';
            } else if (systemState === 'failing') {
                statusDisplay.textContent = 'Equilibrium: FAILING';
                statusDisplay.style.color = '#f80';
            } else if (systemState === 'endangered') {
                statusDisplay.textContent = 'Equilibrium: ENDANGERED';
                statusDisplay.style.color = '#fa0';
            } else {
                statusDisplay.textContent = 'Equilibrium: STABLE';
                statusDisplay.style.color = '#5f5';
            }
        }

        function animate() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            applyCentralForce();
            applySprings();

            for (let i = 0; i < bodies.length; i++) {
                bodies[i].vx *= DAMPING;
                bodies[i].vy *= DAMPING;
                bodies[i].x += bodies[i].vx * speed;
                bodies[i].y += bodies[i].vy * speed;
                bodies[i].trail.push({x: bodies[i].x, y: bodies[i].y});
                if (bodies[i].trail.length > 150) bodies[i].trail.shift();
                const avgDist = getAverageDistance();
                const deformation = Math.min(30, Math.abs(avgDist - NATURAL_LENGTH) / NATURAL_LENGTH * 50);
                bodies[i].radius = bodies[i].originalRadius + deformation;
            }

            if (checkTerminalFailure() && systemState !== 'terminal') {
                systemState = 'terminal';
                updateStatus();
            }

            for (let i = 0; i < bodies.length; i++) {
                for (let j = i + 1; j < bodies.length; j++) {
                    const a = bodies[i], b = bodies[j];
                    const dx = b.x - a.x, dy = b.y - a.y;
                    const dist = Math.sqrt(dx*dx + dy*dy);
                    const tension = Math.max(0, dist - NATURAL_LENGTH) / NATURAL_LENGTH;
                    let r, g, bv;
                    if (systemState === 'terminal') { r=255; g=0; bv=0; }
                    else { r=Math.min(255,100+tension*155); g=Math.min(255,200-tension*150); bv=Math.min(255,100-tension*100); }
                    ctx.strokeStyle = `rgba(${r},${g},${bv},0.6)`;
                    ctx.lineWidth = Math.max(1, 4 - tension * 2);
                    ctx.beginPath();
                    ctx.moveTo(a.x, a.y);
                    ctx.lineTo(b.x, b.y);
                    ctx.stroke();
                }
            }

            for (let i = 0; i < bodies.length; i++) {
                const body = bodies[i];
                if (body.trail.length > 1) {
                    ctx.beginPath();
                    ctx.moveTo(body.trail[0].x, body.trail[0].y);
                    for (let j = 1; j < body.trail.length; j++) {
                        ctx.lineTo(body.trail[j].x, body.trail[j].y);
                    }
                    ctx.strokeStyle = body.color + '30';
                    ctx.lineWidth = body.radius / 3;
                    ctx.stroke();
                }
                ctx.beginPath();
                ctx.arc(body.x, body.y, body.radius, 0, Math.PI * 2);
                if (systemState === 'terminal') ctx.fillStyle = '#f00';
                else if (systemState === 'failing') ctx.fillStyle = '#f80';
                else ctx.fillStyle = body.color;
                ctx.fill();
                ctx.strokeStyle = '#000';
                ctx.lineWidth = 2;
                ctx.stroke();
            }

            ctx.fillStyle = '#aaa';
            ctx.font = '18px sans-serif';
            ctx.textAlign = 'left';
            ctx.fillText(`Disturbances: ${disturbanceCount}`, 20, canvas.height - 30);

            if (systemState === 'terminal') {
                ctx.fillStyle = 'rgba(255,0,0,0.3)';
                ctx.fillRect(0, 0, canvas.width, canvas.height);
                ctx.fillStyle = '#fff';
                ctx.font = '48px sans-serif';
                ctx.textAlign = 'center';
                ctx.fillText('TERMINAL FAILURE', canvas.width / 2, canvas.height / 2);
            }

            animationId = requestAnimationFrame(animate);
        }

        canvas.addEventListener('click', (e) => disturb(e.clientX, e.clientY));
        speedControl.addEventListener('input', (e) => speed = parseFloat(e.target.value));
        resetBtn.addEventListener('click', reset);
        animate();
    </script>
</body>
</html>

---
Equilibrium: Three bodies in equilateral triangle formation, rotating around a center point with spring-like connections maintaining equal distances (~180 units).
Corrective influence: Spring forces between bodies (Hooke's law) and weak central gravitational pull.
Accumulates: Kinetic energy (velocity) that isn't fully dissipated by damping, increasing connection tension and system stress.
Instability visible: Connection color shifts (redder = more tension), connections thin, bodies deform/grow, trails become chaotic, status text updates.
Terminal failure: Any connection stretches beyond 2.2× natural length OR total energy > 800; visualized by red overlay, "TERMINAL FAILURE" text, red connections/body.
Surprise: Chaotic but persistent oscillation patterns emerge after 3-4 disturbances; resonance effects from timed clicks cause wild swings; trail patterns form beautiful spirals; body deformation creates intuitive stress visualization.
